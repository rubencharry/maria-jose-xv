<script data-purpose="audio-toggle">
    // Mock audio play/pause simulation toggle for mobile UX
    const musicBtn = document.getElementById('musicToggleBtn');
    const playIcon = document.getElementById('playIcon');
    const pauseIcon = document.getElementById('pauseIcon');
    let isPlaying = false;

    if (musicBtn) {
      musicBtn.addEventListener('click', () => {
        isPlaying = !isPlaying;
        const iconContainer = musicBtn.querySelector('.music-pulse') || musicBtn.querySelector('div');
        if (isPlaying) {
          playIcon.classList.add('hidden');
          pauseIcon.classList.remove('hidden');
          if (iconContainer) iconContainer.classList.remove('music-pulse');
        } else {
          playIcon.classList.remove('hidden');
          pauseIcon.classList.add('hidden');
          if (iconContainer) iconContainer.classList.add('music-pulse');
        }
      });
    }
  </script>
<script data-purpose="countdown-logic">
    // Live countdown calculation to December 29, 2026
    const targetDate = new Date('December 29, 2026 17:00:00').getTime();

    function updateCountdown() {
      const now = new Date().getTime();
      const difference = targetDate - now;

      if (difference > 0) {
        const days = Math.floor(difference / (1000 * 60 * 60 * 24));
        const hours = Math.floor((difference % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
        const minutes = Math.floor((difference % (1000 * 60 * 60)) / (1000 * 60));
        const seconds = Math.floor((difference % (1000 * 60)) / 1000);

        document.getElementById('daysCount').innerText = days;
        document.getElementById('hoursCount').innerText = hours < 10 ? '0' + hours : hours;
        document.getElementById('minsCount').innerText = minutes < 10 ? '0' + minutes : minutes;
        document.getElementById('secsCount').innerText = seconds < 10 ? '0' + seconds : seconds;
      }
    }

    setInterval(updateCountdown, 1000);
    updateCountdown();
  </script>
<script data-purpose="clipboard-copy">
    // Copy CLABE data with feedback
    const copyBtn = document.getElementById('copyClabeBtn');
    const copyBtnText = document.getElementById('copyBtnText');
    const clabeText = document.getElementById('clabeNumber').innerText;

      navigator.clipboard.writeText(clabeText.replace(/\s+/g, '')).then(() => {
        copyBtnText.innerText = '¡CLABE Copiada con Éxito!';

        setTimeout(() => {
          copyBtnText.innerText = 'Copiar Datos Bancarios';
          copyBtn.classList.remove('bg-powder-blue-200', 'text-powder-blue-500');
        }, 2500);
      }).catch(() => {
        alert('CLABE: ' + clabeText);
      });
    });
  </script>
<script data-purpose="rsvp-handler">
    // WhatsApp RSVP redirection with formatted message
    function handleRSVP(event) {
      event.preventDefault();
      const name = document.getElementById('guestName').value.trim();
      const status = document.getElementById('attendingStatus').value;
      const count = document.getElementById('guestCount').value;
      const message = document.getElementById('guestMessage').value.trim();

      const phone = "5215512345678"; // Replace with organizer/quinceañera WhatsApp phone number

      let text = `¡Hola Ximena! Confirmo mi asistencia para tus XV Años:%0A%0A`;
      text += `*Nombre:* ${encodeURIComponent(name)}%0A`;
      text += `*Respuesta:* ${encodeURIComponent(status)}%0A`;
      text += `*Pases confirmados:* ${encodeURIComponent(count)}%0A`;
      if (message) {
        text += `*Mensaje:* ${encodeURIComponent(message)}%0A`;
      }

      window.open(`https://wa.me/${phone}?text=${text}`, '_blank');
    }
  </script>
